$engine = 'D:/Program/Godot/Standard/Godot_v4.7.2-stable_win64_console.exe'
$exportDir = Join-Path ([IO.Path]::GetTempPath()) ([Guid]::NewGuid().ToString())

    Copy-Item -LiteralPath $file -Destination (Join-Path $exportDir $file)
